  id (BIGINT, PK)
  fecha (DATE)
  trabajador_id (BIGINT, FK a empleados.id)
  planta_presentacion_id (BIGINT, FK a planta_presentacion.id)
  cantidad (INT) // Cantidad descontada del inventario
  detalle (VARCHAR(255)) // Motivo del descargo. Ejemplo: plantas muertas

  // CAMPOS DE REFERENCIA PARA JOINS:

}

entrada_exterior{
  id (BIGINT, PK)
  fecha (DATE)
  trabajador_id (BIGINT, FK a empleados.id)
  planta_presentacion_id (BIGINT, FK a planta_presentacion.id)
  tipo (VARCHAR(255)) // compra/devolucion
  cantidad (INT) // Cantidad agregada al inventario
  detalle (VARCHAR(255)) // Ejemplo: compra de plantas del vivero de soya
  total_precio (DOUBLE, NULLABLE) // Vacío si es devolución
  crbyat (TIMESTAMP)
  upbyat (TIMESTAMP)

  // CAMPOS DE REFERENCIA PARA JOINS:
  // nombre_planta (JOIN por medio de planta_presentacion.planta_id -> plantas.nombre)
  // presentacion (JOIN por medio de planta_presentacion.presentacion_id -> presentaciones.nombre)
  // codigo (JOIN por medio de planta_presentacion.codigo)
  // requisicion (JOIN por medio de presentaciones.requisicion)
}


CREATE VIEW vista_bitacora_ml AS

-- 1. Movimientos de Producción
SELECT 
    id AS movimiento_id,
    'Producción' AS tipo_movimiento,
    fecha,
    strftime('%Y', fecha) AS anio,
    trabajador_id,
    planta_presentacion_id AS planta_presentacion_origen_id,
    NULL AS planta_presentacion_destino_id,
    cantidad,
    'Tiempo siembra: ' || siembra_tiempo_horas || ' horas' AS detalle
FROM produccion

UNION ALL

-- 2. Movimientos de Cambio de Presentación
SELECT 
    id AS movimiento_id,
    'Cambio de Presentación' AS tipo_movimiento,
    fecha,
    strftime('%Y', fecha) AS anio,
    trabajador_id,
    planta_presentacion_origen_id,
    planta_presentacion_destino_id,
    cantidad,
    NULL AS detalle
FROM cambio_presentacion

UNION ALL

-- 3. Movimientos de Descargo
SELECT 
    id AS movimiento_id,
    'Descargo' AS tipo_movimiento,
    fecha,
    strftime('%Y', fecha) AS anio,
    trabajador_id,
    planta_presentacion_id AS planta_presentacion_origen_id,
    NULL AS planta_presentacion_destino_id,
    cantidad,
    detalle
FROM descargo

UNION ALL

-- 4. Movimientos de Entrada Exterior
SELECT 
    id AS movimiento_id,
    'Entrada Exterior (' || tipo || ')' AS tipo_movimiento,
    fecha,
    strftime('%Y', fecha) AS anio,
    trabajador_id,
    NULL AS planta_presentacion_origen_id, 
    planta_presentacion_id AS planta_presentacion_destino_id,
    cantidad,
    detalle
FROM entrada_exterior;
